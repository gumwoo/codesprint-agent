# 커리큘럼/계약 하네스가 쓰는 의존성.
#
# CI 와 로컬이 같은 파일을 보게 한다. 예전에는 README 가 `pip install pyyaml`,
# CI 가 `pip install pyyaml jsonschema` 로 갈라져 있었다 - 새로 clone 한 사람이
# README 대로 따라 하면 jsonschema 가 없어 하네스가 import 에서 죽는다.
# check_curriculum.py 의 deps 검사가 이 드리프트를 막는다.
#
#   pip install -r requirements-dev.txt
#
# 버전을 고정하는 이유는 검사 결과가 실행 환경에 따라 달라지면 안 되기 때문이다.
#
# ⚠️ pyyaml 은 6.0.3 이상이어야 한다. 6.0.2 에는 Python 3.13/3.14 용 wheel 이 없어
#    로컬에서 소스 빌드를 시도하다 컴파일러가 없으면 설치 자체가 실패한다
#    ("Unable to find a compatible Visual Studio installation").
#    CI 는 Python 3.12 로 고정돼 있어 드러나지 않지만, 최신 Python 을 쓰는 사람의
#    로컬만 조용히 막힌다. 실제로 3.14 에서 재현했다.

pyyaml==6.0.3
jsonschema==4.23.0

# 없어 로컬(3.14)에서 설치가 실패한다 - CI(3.12)에서만 되는 핀은 새로 clone 한
# 사람을 막는다.
psycopg[binary]==3.2.13
